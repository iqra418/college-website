Question 1: Enter a number from the user and find out the binary of that number.

Code

#include <stdio.h>

int main()
{
    int num, rem, arr[16], x = 0, i;
    printf("Enter the number to find its binary : ");
    scanf("%d", &num);

    while (num != 0)
    {
        rem = num % 2;

        arr[x] = rem;
        num = num / 2;
        x++;
    }

    for (i = x - 1; i >= 0; i--)
    {
        printf("%d", arr[i]);
    }
}

    printf("Enter a number to check if it is a palindrome or not: ");

Question 24: Enter a number from user and check it is Armstrong number or not.

Code

#include <stdio.h>

int main() {
    int num, rem, sum = 0, org;

    printf("Enter a number: ");
    scanf("%d", &num);

    org = num; 

    while (num != 0) {
        rem = num % 10;          
        sum = sum + (rem * rem * rem); 
        num = num / 10;          
    }

    if (sum == org)
        printf("%d is an Armstrong number.\n", org);
    else
        printf("%d is not an Armstrong number.\n",org);

    return 0;
}

Output

Enter a number: 52
52 is not an Armstrong number.


Enter a number: 370
370 is an Armstrong number.
