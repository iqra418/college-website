Question 2: Enter the filename from the user and enter the data from the user and put in the file but data should be in capital letter.

Code

#include<conio.h>
#include<dos.h>

{
FILE *f;

int i;

}

}

Enter data:mywebsite

Question 3: Enter the filename from the user and enter the data from the user till the user want to enter the data.

Code

#include<stdio.h>
#include<conio.h>
#include<dos.h>
void main()
{
FILE *f;
char fname[20], data[20], ch;
clrscr();
printf("Enter filename:");
scanf("%s",fname);

f=fopen(fname,"w");
do{
printf("Enter data:");
scanf("%s",data);
fprintf(f,"%s",data);
printf("Do u want to enter more y/n:");
scanf(" %c",&ch);
}while(ch=='y' || ch=='Y');
fclose(f);
delay(3000);
}

Do u want to enter more y/n: y

Enter data: World
Do u want to enter more y/n: y

Question 4: Enter the filename from the user and enter the data from the user and copy that file into another file.

Code

#include<stdio.h>
#include<conio.h>
#include<dos.h>
void main()
{
FILE *f1, *f2;
char fname1[20], fname2[20], data[100];
int ch;
clrscr();
printf("Enter filename:");
scanf("%s",fname1);

f1=fopen(fname1,"w");
printf("Enter data:");
scanf("%s",data);

fprintf(f1,"%s",data);
fclose(f1);

printf("Enter second filename:");
scanf("%s",fname2);

f1=fopen(fname1,"r");
f2=fopen(fname2,"w");

while((ch=fgetc(f1))!=EOF)
{
fputc(ch,f2);
}
printf("File copied successfully");
fclose(f1);
fclose(f2);
delay(3000);
}

Output

Enter filename:file4
Enter data:bca
Enter second filename:file5
File copied successfully

#include<conio.h>

{
