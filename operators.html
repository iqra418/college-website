Question 11: Enter marks of students and calculate average of student’s marks.

Code

#include <stdio.h>
void main() {
    int n, i, marks, sum=0;
    printf("Enter number of students: ");
    scanf("%d", &n);
    for(i=1;i<=n;i++){
        printf("Enter marks of student %d: ", i);
        scanf("%d",&marks);
        sum += marks;
    }
    printf("Average marks: %.2f", (float)sum/n);
}

Output

Enter number of students: 5
Enter marks of student 1: 55
Enter marks of student 2: 97
Enter marks of student 3: 75
Enter marks of student 4: 43
Enter marks of student 5: 20
Average marks: 58.00
Question 12: Calculate simple interest = (P x R x T)/ 100.

Code

#include <stdio.h>
void main() {
    float P, R, T, SI;
    printf("Enter Principal, Rate, Time: ");
    scanf("%f %f %f",&P,&R,&T);
    SI = (P*R*T)/100;
    printf("Simple Interest: %.2f", SI);
}

Output

Enter Principal, Rate, Time: 52 81 98
Simple Interest: 4127.76
